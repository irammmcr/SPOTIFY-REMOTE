// Obtener qué DJ estamos viendo desde la URL (?dj=iram) o del localStorage const urlParams = new URLSearchParams(window.location.search); let currentDJ = urlParams.get('dj') || localStorage.getItem('remotify_dj') || 'iram'; // Actualizar el título principal document.querySelector("h1, .header, title").innerText = `DJ ${currentDJ.toUpperCase()}`; async function updateState() { try { const res = await fetch(`/api/dj/${currentDJ}/state`); const data = await res.json(); // Actualizar Canción Actual const now = data.nowPlaying; if (now && now.name !== "Offline") { document.getElementById("nowPlaying").innerText = `${now.name} • ${now.artist} (${now.user})`; } else { document.getElementById("nowPlaying").innerText = "Sin música activa en Spotify"; } // Actualizar Cola const queueEl = document.getElementById("queue"); if (queueEl) { queueEl.innerHTML = (data.queue || []).map(item => `
${item.name} • ${item.user}
`).join("") || "Cola vacía"; } // Actualizar Historial const historyEl = document.getElementById("history"); if (historyEl) { historyEl.innerHTML = (data.history || []).map(item => `
${item.name} • ${item.user}
`).join("") || "Sin historial"; } } catch (e) { console.error("Error cargando estado:", e); } } async function sendTrack(mode) { const song = document.getElementById("songInput").value; const user = document.getElementById("userInput").value || "Anónimo"; if (!song) return; const res = await fetch(`/api/dj/${currentDJ}/search?q=${encodeURIComponent(song)}&user=${encodeURIComponent(user)}&mode=${mode}`); const data = await res.json(); if (data.ok) { document.getElementById("songInput").value = ""; updateState(); } else { alert("No se encontró la canción o el DJ no tiene Spotify conectado."); } } // Escuchar los clicks de los botones (Asegúrate de que los IDs coincidan con tu HTML) // En tus botones puedes poner onclick="sendTrack('queue')" y onclick="sendTrack('now')" setInterval(updateState, 3000); updateState();
